-- Mini PC PostgreSQL setup for drive
-- Run as a PostgreSQL superuser, for example:
--   psql -U postgres -f psql.txt
--
-- Current app config expects:
--   database: drive_db
--   username: postgres
--   password: 12341234

-- 1. Create database if it does not exist.
SELECT 'CREATE DATABASE drive_db'
WHERE NOT EXISTS (
  SELECT FROM pg_database WHERE datname = 'drive_db'
)\gexec

-- 2. Connect to the application database.
\connect drive_db

-- 3. Create users table.
CREATE TABLE IF NOT EXISTS users (
  id BIGSERIAL PRIMARY KEY,
  username VARCHAR(50) NOT NULL,
  password VARCHAR(255) NOT NULL,
  role VARCHAR(20),
  CONSTRAINT uk_users_username UNIQUE (username)
);

-- 4. Create photo folders table.
CREATE TABLE IF NOT EXISTS photo_folders (
  id BIGSERIAL PRIMARY KEY,
  owner_id VARCHAR(100) NOT NULL,
  folder_path VARCHAR(300) NOT NULL,
  created_at TIMESTAMP NOT NULL,
  updated_at TIMESTAMP,
  sort_order INTEGER,
  CONSTRAINT uk_photo_folders_owner_folder UNIQUE (owner_id, folder_path)
);

-- 5. Create photos table.
CREATE TABLE IF NOT EXISTS photos (
  id BIGSERIAL PRIMARY KEY,
  owner_id VARCHAR(100) NOT NULL,
  original_name VARCHAR(255) NOT NULL,
  storage_key VARCHAR(500) NOT NULL,
  thumbnail_storage_key VARCHAR(500),
  content_type VARCHAR(255) NOT NULL,
  file_size BIGINT NOT NULL,
  created_at TIMESTAMP NOT NULL,
  deleted_at TIMESTAMP,
  folder_path VARCHAR(300) NOT NULL,
  width INTEGER,
  height INTEGER,
  taken_at TIMESTAMP,
  camera_make VARCHAR(100),
  camera_model VARCHAR(200),
  focal_length VARCHAR(50),
  f_number VARCHAR(50),
  exposure_time VARCHAR(50),
  iso INTEGER,
  lens_model VARCHAR(200),
  tags VARCHAR(1000),
  CONSTRAINT uk_photos_storage_key UNIQUE (storage_key)
);

-- 6. Migration safety for existing Mini PC databases.
-- If the database already existed before thumbnail support, this adds the new column.
ALTER TABLE photos
  ADD COLUMN IF NOT EXISTS thumbnail_storage_key VARCHAR(500);

ALTER TABLE photos
  ADD COLUMN IF NOT EXISTS deleted_at TIMESTAMP;

-- 7. Helpful indexes for gallery/folder queries.
CREATE INDEX IF NOT EXISTS idx_photos_owner_folder
  ON photos (owner_id, folder_path);

CREATE INDEX IF NOT EXISTS idx_photos_owner_folder_active
  ON photos (owner_id, folder_path)
  WHERE deleted_at IS NULL;

CREATE INDEX IF NOT EXISTS idx_photos_owner_deleted
  ON photos (owner_id, deleted_at);

CREATE INDEX IF NOT EXISTS idx_photos_owner
  ON photos (owner_id);

CREATE INDEX IF NOT EXISTS idx_photos_created_at
  ON photos (created_at);

CREATE INDEX IF NOT EXISTS idx_photos_taken_at
  ON photos (taken_at);

CREATE INDEX IF NOT EXISTS idx_photo_folders_owner
  ON photo_folders (owner_id);

CREATE INDEX IF NOT EXISTS idx_photo_folders_owner_sort
  ON photo_folders (owner_id, sort_order);

-- 8. Verify created tables.
\dt
